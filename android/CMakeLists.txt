cmake_minimum_required(VERSION 3.24)
project(dasrmlui_android LANGUAGES C CXX)
if(NOT ANDROID)
  message(FATAL_ERROR "Use the Android NDK toolchain (see docs/android.md)")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(DASSDL3_DIR "${ROOT}/../dasSDL3" CACHE PATH "Android-capable dasSDL3 checkout")
set(DAS "${DASSDL3_DIR}/third_party/daScript")
set(BUILD_TESTING OFF)
set(DAS_LIVE_HOST_DISABLED OFF)
add_subdirectory("${DASSDL3_DIR}" dassdl3 EXCLUDE_FROM_ALL)

include(FetchContent)
# Use the same RmlUi internals and FreeType revision as the Web host.
file(READ "${ROOT}/web/dependencies.json" pins)
foreach(dep RmlUi freetype)
  string(JSON repo GET "${pins}" "${dep}" url)
  string(JSON rev GET "${pins}" "${dep}" revision)
  FetchContent_Declare(${dep} GIT_REPOSITORY "${repo}" GIT_TAG "${rev}")
endforeach()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(dep ZLIB BZIP2 PNG HARFBUZZ BROTLI)
  set(FT_DISABLE_${dep} ON CACHE BOOL "" FORCE)
endforeach()
FetchContent_MakeAvailable(freetype)
if(NOT TARGET Freetype::Freetype)
  add_library(Freetype::Freetype ALIAS freetype)
endif()
set(FREETYPE_INCLUDE_DIR_ft2build "${freetype_SOURCE_DIR}/include" CACHE PATH "")
set(FREETYPE_INCLUDE_DIR_freetype2 "${freetype_SOURCE_DIR}/include" CACHE PATH "")
set(FREETYPE_LIBRARY freetype CACHE STRING "")
include("${ROOT}/cmake/vector_plugins.cmake")
# The export map is for rlottie's own shared object. A static dependency
# must not hide SDL_main and the JNI entry points of the application.
get_target_property(lottie_links rlottie INTERFACE_LINK_LIBRARIES)
if(lottie_links)
  list(FILTER lottie_links EXCLUDE REGEX "version-script")
  set_property(TARGET rlottie PROPERTY INTERFACE_LINK_LIBRARIES "${lottie_links}")
endif()
set(RMLUI_SAMPLES OFF CACHE BOOL "" FORCE)
set(RMLUI_THIRDPARTY_CONTAINERS OFF CACHE BOOL "" FORCE)
set(RMLUI_PRECOMPILED_HEADERS OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(RmlUi)
add_library(main SHARED "${ROOT}/src/host.cpp" "${ROOT}/src/live_host.cpp" "${ROOT}/src/rml_module.cpp"
  "${ROOT}/src/gl3/RmlUi_Renderer_GL3.cpp" "${rmlui_SOURCE_DIR}/Backends/RmlUi_Platform_SDL.cpp")
target_include_directories(main PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${DASSDL3_DIR}/mobile" "${DAS}/modules/dasLiveHost/src"
  "${rmlui_SOURCE_DIR}/Source/Core" "${rmlui_SOURCE_DIR}/Backends")
target_compile_definitions(main PRIVATE RMLUI_SDL_VERSION_MAJOR=3 DAS_NO_FILEIO=1)
target_link_libraries(main PRIVATE dasSDL3 libDasModuleLiveHost RmlUi::RmlUi RmlUi::Debugger GLESv3)

FetchContent_GetProperties(SDL3)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/sdl-gradle-wrapper.txt" "${sdl3_SOURCE_DIR}/android-project/gradlew.bat\n")
set(jni "jniLibs/${CMAKE_ANDROID_ARCH_ABI}")
install(TARGETS main SDL3-shared LIBRARY DESTINATION "${jni}" COMPONENT apk)
if(ANDROID_STL STREQUAL "c++_shared")
  install(FILES "${CMAKE_SYSROOT}/usr/lib/${CMAKE_LIBRARY_ARCHITECTURE}/libc++_shared.so"
    DESTINATION "${jni}" COMPONENT apk)
endif()
install(DIRECTORY "${sdl3_SOURCE_DIR}/android-project/app/src/main/java/" DESTINATION java COMPONENT apk)
foreach(path daslib modules/dasGlsl/glsl modules/dasLiveHost/live)
  install(DIRECTORY "${DAS}/${path}/" DESTINATION "runtime/daScript/${path}" COMPONENT apk
    FILES_MATCHING PATTERN "*.das" PATTERN "_aot_generated" EXCLUDE)
endforeach()
foreach(path "${DASSDL3_DIR}/dassdl3" "${ROOT}/dasrmlui")
  install(DIRECTORY "${path}" DESTINATION runtime COMPONENT apk FILES_MATCHING PATTERN "*.das")
endforeach()
install(FILES "${ROOT}/tests/android_font_input.das" DESTINATION runtime/examples COMPONENT apk)
install(DIRECTORY "${ROOT}/examples/" DESTINATION runtime/examples COMPONENT apk
  FILES_MATCHING PATTERN "*.das" PATTERN "*.svg" PATTERN "*.json" PATTERN "*.png")
install(DIRECTORY "${rmlui_SOURCE_DIR}/Samples/assets/" DESTINATION runtime/assets COMPONENT apk
  FILES_MATCHING PATTERN "LatoLatin-*.ttf" PATTERN "LICENSE.txt")
install(FILES "${ROOT}/web/assets/DejaVuSans.ttf" "${ROOT}/web/assets/LICENSE-DejaVuSans.ttf.txt"
  DESTINATION runtime/web/assets COMPONENT apk)
foreach(dep SDL daScript)
  install(DIRECTORY "${DASSDL3_DIR}/licenses/${dep}/" DESTINATION "runtime/licenses/${dep}" COMPONENT apk)
endforeach()
install(FILES "${DASSDL3_DIR}/LICENSE" DESTINATION runtime/licenses/dasSDL3 COMPONENT apk)
install(FILES "${ROOT}/LICENSE" DESTINATION runtime/licenses/dasrmlui COMPONENT apk)
install(FILES "${rmlui_SOURCE_DIR}/LICENSE.txt" DESTINATION runtime/licenses/RmlUi COMPONENT apk)
install(FILES "${freetype_SOURCE_DIR}/docs/FTL.TXT" DESTINATION runtime/licenses/FreeType COMPONENT apk)
install(FILES "${lunasvg_SOURCE_DIR}/LICENSE" DESTINATION runtime/licenses/lunasvg COMPONENT apk)
install(FILES "${lunasvg_SOURCE_DIR}/plutovg/LICENSE" DESTINATION runtime/licenses/plutovg COMPONENT apk)
install(FILES "${rlottie_SOURCE_DIR}/COPYING" DESTINATION runtime/licenses/rlottie COMPONENT apk)
install(DIRECTORY "${rlottie_SOURCE_DIR}/licenses/" DESTINATION runtime/licenses/rlottie COMPONENT apk)
